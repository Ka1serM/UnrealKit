cmake_minimum_required(VERSION 3.29)
project(UnrealKit LANGUAGES C CXX)

include(cmake/CompilerOptions.cmake)

if(EMSCRIPTEN)
    set(unrealkit_platform Web)
elseif(WIN32)
    set(unrealkit_platform Windows)
else()
    set(unrealkit_platform Linux)
endif()
foreach(platform Windows Linux Web)
    string(TOUPPER "${platform}" platform_name)
    if(unrealkit_platform STREQUAL platform)
        set(PLATFORM_${platform_name} ON)
    else()
        set(PLATFORM_${platform_name} OFF)
    endif()
endforeach()

add_subdirectory(external)
add_subdirectory(schema)
add_subdirectory(src)
